 insira(celula *p, int y){

 void   nova->conteudo = y;

    nova->seg->ant = nova;

    return;

    int num = 0;
    if (p->seg == p && p->ant == p)
        printf("Stack Underflow");
    else{
        celula *last_celula = p->seg;

        p->seg = last_celula->seg;
        last_celula->seg->ant = p;
        
        free(last_celula);
    }
    
    return num;

#include <stdio.h>
#include <stdlib.h>

#define N 30

struct cell
{
    int conteudo;
    struct cell *seg, *ant;
};
typedef struct cell celula;

void insira(celula **p, int y);
int desempilhe(celula **p);

int main()
{
    celula *lst = NULL;

    insira(&lst, 33);
    insira(&lst, 7);
    insira(&lst, 11);
    printf("%d - ", desempilhe(&lst));
    printf("%d - ", desempilhe(&lst));
    insira(&lst, 2);
    printf("%d - ", desempilhe(&lst));

    return 0;
}

void insira(celula **p, int y)
{
    celula *nova = (celula *)malloc(sizeof(celula));
    nova->conteudo = y;
    nova->ant = NULL;
    if (p == NULL){
        nova->seg = NULL;
        (*p)->ant = nova;
    }
    else
        nova->seg = *p;

    *p = nova;
    return;
}
int desempilhe(celula **p)
{
    int num = 0;

    if (p == NULL){
        printf("Stack Underflow");
        num = -1;
    }
    else
    {
        celula *topo = (*p);
        num = topo->conteudo;
        *p = topo->seg;
        if(*p != NULL)
            (*p)->ant = NULL;

        free(topo);
    }

    return num;
}
